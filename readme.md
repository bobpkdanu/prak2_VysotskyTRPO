git init - создание локального репозитория
git commit - создание коммита
git branch - создание ветки
git switch - смена ветки
git merge - слияние веток
git diff - сравнение веток 
git fetch - получение изменений в удаленном репозитории
git push - отправка ветки в удалённый репозиторий
git remote - связь удалённого репозитория с локальным
git add - добавление файла в репозиторий
git status - проверка статуса
